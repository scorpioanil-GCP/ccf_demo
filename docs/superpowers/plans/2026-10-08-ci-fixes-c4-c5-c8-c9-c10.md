``tmp_path``. Offline: git only, no estate.

from __future__ import annotations

import pathlib
import subprocess

import pytest

from tests.examples.specimen_base import branch_point


def _git(cwd: pathlib.Path, *args: str) -> str:
    return subprocess.run(
        ["git", *args], cwd=cwd, capture_output=True, text=True, check=True
    ).stdout.strip()


def _commit(repo: pathlib.Path, name: str) -> str:
    (repo / name).write_text(name, encoding="utf-8")
    _git(repo, "add", name)
    _git(
        repo,
        "-c", "user.name=t", "-c", "user.email=t@t", "-c", "commit.gpgsign=false",
        "commit", "-q", "-m", name,
    )
    return _git(repo, "rev-parse", "HEAD")


@pytest.fixture
def origin(tmp_path: pathlib.Path) -> tuple[pathlib.Path, str]:
    """A repository whose ``main`` holds one commit: the branch point."""
    repo = tmp_path / "origin"
    repo.mkdir()
    _git(repo, "init", "-q", "-b", "main")
    return repo, _commit(repo, "base")


def _hosted_branch_checkout(origin: pathlib.Path, dest: pathlib.Path) -> pathlib.Path:
    """The shape actions/checkout leaves: remote-tracking refs, one local branch."""
    _git(origin, "checkout", "-q", "-b", "lane")
    _commit(origin, "lane-work")
    _git(origin, "checkout", "-q", "main")
    dest.mkdir()
    _git(dest, "init", "-q")
    _git(dest, "remote", "add", "origin", str(origin))
    _git(dest, "fetch", "-q", "origin", "+refs/heads/*:refs/remotes/origin/*")
    _git(dest, "checkout", "-q", "--force", "-B", "lane", "refs/remotes/origin/lane")
    return dest


def test_a_hosted_branch_checkout_measures_from_origin_main(
    origin: tuple[pathlib.Path, str], tmp_path: pathlib.Path
) -> None:
    repo, base = origin
    checkout = _hosted_branch_checkout(repo, tmp_path / "hosted")
    no_local_main = subprocess.run(
        ["git", "rev-parse", "--verify", "--quiet", "main"], cwd=checkout, capture_output=True
    )
    assert no_local_main.returncode != 0, "the fixture must have no local main, as CI has none"
    assert branch_point(checkout) == base


def test_a_local_main_is_used_when_present(
    origin: tuple[pathlib.Path, str],
) -> None:
    repo, base = origin
    _git(repo, "checkout", "-q", "-b", "lane")
    _commit(repo, "lane-work")
    assert branch_point(repo) == base

```python
"""Where a frozen-specimen pin measures from (D-IMPL-378 C9).

``D-IMPL-261`` freezes specimens, and each pin diffs its specimen against the
commit its branch left ``main`` at. A local clone has a ``main`` branch. A
hosted branch checkout does not: ``actions/checkout`` fetches every branch as
``refs/remotes/origin/*`` and makes only the checked-out one local, so the bare
``git merge-base HEAD main`` exits 128 there (the D-IMPL-378 CI diagnosis,
§4.2). ``origin/main`` is the same commit under another name.
"""

from __future__ import annotations

import pathlib
import subprocess

#: Tried in order. A ref that resolves is used and never second-guessed.
_CANDIDATES = ("main", "origin/main")


def _resolves(repo_root: pathlib.Path, ref: str) -> bool:
    return (
        subprocess.run(
            ["git", "rev-parse", "--verify", "--quiet", f"{ref}^{{commit}}"],
            cwd=repo_root,
            capture_output=True,
        ).returncode
        == 0
    )


def branch_point(repo_root: pathlib.Path) -> str:
    """The merge base of ``HEAD`` with ``main``, or with ``origin/main`` where no
    local ``main`` exists.

    Only a ref that does not resolve moves the lookup on. A ``main`` that
    resolves but shares no history with ``HEAD`` raises
    ``subprocess.CalledProcessError`` rather than trying ``origin/main``: a pin
    must never pass by measuring from the wrong place. When neither resolves,
    ``AssertionError`` names both.
    """
    for ref in _CANDIDATES:
        if _resolves(repo_root, ref):
            return subprocess.run(
                ["git", "merge-base", "HEAD", ref],
                cwd=repo_root,
                capture_output=True,
                text=True,
                check=True,
            ).stdout.strip()
    raise AssertionError(
        "neither `main` nor `origin/main` resolves in this checkout, so the frozen"
        " specimen cannot be measured against its branch point"
    )
```

import pathlib

REPO_ROOT = pathlib.Path(__file__).resolve().parents[2]

def _xfail_keywords(function: ast.FunctionDef) -> dict[str, ast.expr] | None:
    for decorator in function.decorator_list:
        if isinstance(decorator, ast.Call) and ast.unparse(decorator.func) == "pytest.mark.xfail":
            return {keyword.arg: keyword.value for keyword in decorator.keywords if keyword.arg}
    return None


def test_the_roster_is_the_twelve_the_hosted_runs_measured() -> None:
    assert len(_CASES) == 12

    ("module", "name", "expected"),
    _CASES,
    ids=[f"{module.rsplit('/', 1)[-1]}::{name}" for module, name, _ in _CASES],
)
def test_each_frozen_family_red_is_a_strict_expected_failure_naming_its_exception(
    module: str, name: str, expected: str
) -> None:
    tree = ast.parse((REPO_ROOT / module).read_text(encoding="utf-8"))
    functions = {node.name: node for node in tree.body if isinstance(node, ast.FunctionDef)}

    assert "raises" in keywords, f"{module}::{name} names no exception, so it would absorb any failure"
    assert ast.unparse(keywords["raises"]).rsplit(".", 1)[-1] == expected
