## Criterion 1 — the fence is a roster judged by SET EQUALITY

**Instrument.** A small inline Python snippet (recorded verbatim, not a `tools/` script — tooling is
out of scope): (a) reads `ledger_ids:` from every `24.5-NN-PLAN.md` frontmatter; (b) reads
`git log --format=%s phase-24.5-base..HEAD`, keeps ONLY subjects matching
`^(feat|test|fix|refactor)\(24\.5-[0-9]{2}\)`, and greps `24.5-L-[0-9]{2}` from those SUBJECT lines;
(c) unions (a) and (b) and compares by set equality to Wave 1 ∪ Wave 2 ∪ {`L-18`, `L-49`} — 23 ids.
Arrivals and departures are reported separately.

Why the SUBJECT and never the body: a commit body legitimately cites rows the commit does not work
(plan 16's rewrite commit cites `24.5-L-37`, the estate fact `CFT-G15` guards; plan 01's commit cites
`24.5-L-45`, the ruling row it executes); a subject names only the rows the commit works. The body
form is run below as the REJECTED instrument and its false arrivals are recorded.

Command: `.venv/bin/python <scratchpad>/crit1.py` — the snippet:

```python
"""Criterion 1 — the ledger-id union, by SET EQUALITY, with a planted control."""
import glob, re, subprocess, sys
ROOT = "/home/parkave/ccf_code"
PH = f"{ROOT}/.planning/phases/24.5-sdk-gap-catch-up-cp-00-through-cp-04"
WAVE1 = {"L-01","L-02","L-03","L-04","L-05","L-06","L-08","L-09","L-10","L-11","L-12","L-13","L-16","L-21","L-22","L-23","L-27","L-28"}
WAVE2 = {"L-14","L-24","L-26"}
BOOK = {"L-18","L-49"}
RULED = WAVE1 | WAVE2 | BOOK
assert len(RULED) == 23, len(RULED)

def plan_ids():
    out = {}
    for p in sorted(glob.glob(f"{PH}/24.5-[0-9][0-9]-PLAN.md")):
        fm = open(p).read().split("---")[1]
        m = re.search(r"^ledger_ids:\s*\[(.*?)\]", fm, re.M)
        ids = set(re.findall(r"24\.5-(L-\d{2})", m.group(1))) if m else set()
        out[p.rsplit("/",1)[1]] = ids
    return out

SUBJ_RE = re.compile(r"^(feat|test|fix|refactor)\(24\.5-\d{2}\)")
def commit_ids(fmt):
    log = subprocess.run(["git","-C",ROOT,"log",f"--format=%H%x00{fmt}%x01","phase-24.5-base..HEAD"],capture_output=True,text=True,check=True).stdout
    found = {}
    for rec in log.split("\x01"):
        if "\x00" not in rec: continue
        sha, text = rec.strip("\n").split("\x00",1)
        first = text.split("\n",1)[0]
        if not SUBJ_RE.match(first): continue
        ids = set(re.findall(r"24\.5-(L-\d{2})", text))
        if ids: found[sha[:8]] = ids
    return found

def compare(label, claimed):
    arrivals = sorted(claimed - RULED); departures = sorted(RULED - claimed)
    print(f"[{label}] claimed={len(claimed)} ruled={len(RULED)} arrivals={arrivals} departures={departures} -> {'SET-EQUAL' if not arrivals and not departures else 'NOT EQUAL'}")

plans = plan_ids()
print("== (a) ledger_ids: per plan frontmatter")
for k,v in plans.items(): print(f"  {k}: {sorted(v) if v else '[]'}")
A = set().union(*plans.values())
print(f"  union(a) = {sorted(A)}  ({len(A)})")
outside = {k: sorted(v - RULED) for k,v in plans.items() if v - RULED}
print(f"  plans naming an id OUTSIDE the union: {outside if outside else 'none'}")

print("== (b) SUBJECT-line ids of feat|test|fix|refactor(24.5-NN) commits (%s)")
subj = commit_ids("%s")
B = set().union(*subj.values()) if subj else set()
for sha,ids in subj.items(): print(f"  {sha}: {sorted(ids)}")
print(f"  union(b) = {sorted(B)}  ({len(B)})")

print("== (c) union (a) ∪ (b) vs Wave 1 ∪ Wave 2 ∪ {L-18, L-49}")
compare("a∪b", A | B)
print(f"  ids in (a) not in (b): {sorted(A-B)}   ids in (b) not in (a): {sorted(B-A)}")

print("== POSITIVE CONTROL: L-29 planted into the claimed set")
compare("a∪b + planted L-29", (A | B) | {"L-29"})

print("== NEGATIVE CONTROL (REJECTED instrument): the same grep over commit BODIES (%B)")
body = commit_ids("%B")
Bb = set().union(*body.values()) if body else set()
compare("a ∪ body-ids", A | Bb)
for sha,ids in body.items():
    extra = ids - RULED
    if extra: print(f"  false arrival from body of {sha}: {sorted(extra)}")
```
