**The compatibility keyword** (`::run_gates` signature and the `source_attestation` docstring — this is the voice for a new `None`-defaulted input):

```python
def run_gates(
    candidate: ContextProductContract,
    *,
    engine_resolved: frozenset[str],
    now: datetime,
    registry: ControlPlaneProtocol,
    product_class: RegisteredProductClass,
    source_attestation: SourceAttestation | None = None,
) -> ServableProduct | GateRefusal:
    """...
        source_attestation: ... A keyword-only argument with a ``None`` default
            rather than a field on the candidate, because a product does not
            attest its own source; and ``None`` passes — the compatibility
            posture :func:`refuse_attestation` documents — so every
            pre-attestation call site is byte-for-byte unchanged in behavior.
    """
    declared = candidate.lineage.operators_invoked
    empty = _refuse_empty_operators(candidate)
    if empty is not None:
        return empty
    if declared != engine_resolved:
        return LineageMismatchRefusal(...)
    ttl = _refuse_ttl(candidate, now=now, registry=registry, product_class=product_class)
    if ttl is not None:
        return ttl
    attestation = refuse_attestation(attestation=source_attestation, product_class=product_class)
    if attestation is not None:
        return attestation
```
